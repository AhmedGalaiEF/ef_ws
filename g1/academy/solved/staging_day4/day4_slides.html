Knowledge — ArUco detection and why validation matters

  • aruco_pose() always uses the first entry in corners/ids — it does not filter by marker ID (see aruco_setup.html). Keep pickup/dropdown markers out of the same frame, or you get whichever one OpenCV happened to list first.
  • Perception output must never drive motion unchecked: reject a stale frame (timestamp older than max_age_s) and reject a low-confidence detection instead of quietly acting on a guess.

Solution Task 12

Solution — validated_target's two refusals

def validated_target(pipeline, frame, marker_length_m, min_confidence=0.5,
                       max_age_s=1.0, prompt="the delivery package"):
    if time.time() - frame.get("timestamp", time.time()) > max_age_s:
        raise RuntimeError("RGB-D frame is stale.")                       # refusal #1
    rgb = cv2.imdecode(np.frombuffer(frame["rgb_jpeg"], dtype=np.uint8), cv2.IMREAD_COLOR)
    detection = pipeline.detect_object(frame["rgb_jpeg"], prompt)
    if detection.confidence < min_confidence:
        raise RuntimeError(f"Low-confidence detection ({detection.confidence}); refusing to act.")  # refusal #2
    marker_pose = pipeline.aruco_pose(rgb, marker_length_m)
    return pipeline.marker_to_palm_target(marker_pose)

def deliver(pickup_point, dropdown_point, side="right", marker_length_m=0.04, grasp_threshold=0.5):
    robot = _robot
    robot.extend_arm_forward(side=side)                                   # 1  [Task 8]
    robot.gradual_open(side=side)                                         # 2  [Task 11]

    frame = get_rgbd()                                                    # 3  [Task 12]
    detection = robot.pipeline.detect_object(frame["rgb_jpeg"], "the delivery package")
    if detection.confidence < 0.5:
        raise RuntimeError(f"Low-confidence detection ({detection.confidence}); aborting.")
    marker_pose = robot.pipeline.aruco_pose(decode(frame), marker_length_m)
    pickup_target = robot.pipeline.marker_to_palm_target(marker_pose)

    robot.pipeline.execute_incremental_ik(                                # 4  [Task 10]
        robot.ik_increment, robot.current_palm_xyz(side), pickup_target, side=side)
    robot.gradual_close(grasp_threshold, side=side)                       # 5  [Task 11]
    robot.interpolate_to_ll_pose("stable_hold_pose", duration_s=3.0)      # 6  [Task 8]
    robot.navigate_to_point(dropdown_point)                               # 7  [Task 7]

    dropdown_target = robot.pipeline.marker_to_palm_target(               # 8  [Task 8/10/11]
        robot.pipeline.aruco_pose(decode(get_rgbd()), marker_length_m))
    robot.pipeline.execute_incremental_ik(robot.ik_increment, robot.current_palm_xyz(side), dropdown_target, side=side)
    robot.gradual_open(side=side)
    robot.extend_arm_forward(side=side)
    robot.release_arms()
    return {"pickup": pickup_point, "dropdown": dropdown_point}
