§1Knowledge — the SLAM RPC API ids

SlamRpc is a unitree_sdk2py.rpc.client.Client registered against five documented API ids on the "slam_operate" service.

API idOperation
1801start mapping
1802 / 1901stop mapping (with / without a save path)
1804relocate / init-pose
1102single-pose navigation

An all-zero parsed pose means "no valid pose yet", not "robot at the origin" — treat it as absent, the same convention sdk_wrapper._parse_slam_pose uses. (Error codes worth knowing if you see them: 501 = lacking lidar/IMU data; 509 = relocation confidence too low.)

§2Knowledge — SLAM lifecycle and the map-visualization gap

Once a map is saved on the mainboard it may no longer be accessible for visualization — so view_map() does not read the saved file. Instead, capture_map_snapshot() saves a 2-D scatter PNG from the live point-cloud topics while mapping is running, and view_map() just returns the newest one.

Lifecycle: start_mapping() → move robot, capture snapshots → stop_mapping() (saves to mainboard) → relocate() (needs a fresh pose) → add_point(name) ×N → navigate_to_point(name). Named points only make sense after a successful relocate — never persist a point captured before localization is valid.

navigate_path deliberately refuses a silent point-by-point fallback: pass a verified native_path_callback once one exists for this SDK/firmware, or it raises (see academy/todo.txt) — this is a documented limitation, not something to quietly work around.

class SlamRpc(Client):
    def __init__(self):
        super().__init__("slam_operate", False)
        for api_id in (1801, 1802, 1804, 1102, 1901):
            self._RegistApi(api_id, 0)
        self._SetApiVerson("1.0.0.1")

    def _call_json(self, api_id, payload):
        code, data = self._Call(api_id, json.dumps(payload))
        return int(code), data

    def start_mapping(self, slam_type="indoor"):
        return self._call_json(1801, {"data": {"slam_type": slam_type}})

    def stop_mapping(self, save_path=None):
        return self._call_json(1802, {"data": {"address": save_path}}) if save_path else self._call_json(1901, {"data": {}})

    def init_pose(self, x, y, yaw=0.0, address=""):
        qz, qw = math.sin(yaw / 2), math.cos(yaw / 2)
        return self._call_json(1804, {"data": {"x": x, "y": y, "z": 0.0, "q_x": 0.0, "q_y": 0.0, "q_z": qz, "q_w": qw, "address": address}})

    def pose_nav(self, x, y, yaw=0.0):
        qz, qw = math.sin(yaw / 2), math.cos(yaw / 2)
        return self._call_json(1102, {"data": {"targetPose": {"x": x, "y": y, "z": 0.0, "q_x": 0.0, "q_y": 0.0, "q_z": qz, "q_w": qw}, "mode": 1}})

slam_rpc = SlamRpc()
slam_rpc.SetTimeout(10.0)

class Latest:                        # Task 2's pattern, repeated
    def __init__(self, topic, message_type, queue_len=10):
        self.message = None; self.timestamp = 0.0
        self.subscriber = ChannelSubscriber(topic, message_type)
        self.subscriber.Init(self._callback, queue_len)
    def _callback(self, message):
        self.message = message; self.timestamp = time.time()

slam_info_sub = Latest("rt/slam_info", String_)
slam_key_sub = Latest("rt/slam_key_info", String_)

def current_pose():
    for sub in (slam_info_sub, slam_key_sub):
        if sub.message is None:
            continue
        try:
            cur = json.loads(sub.message.data)["data"]["currentPose"]
            x, y = float(cur["x"]), float(cur["y"])
            qz, qw = float(cur.get("q_z", 0.0)), float(cur.get("q_w", 1.0))
            yaw = math.atan2(2*qw*qz, 1 - 2*qz*qz)
        except Exception:
            continue
        if abs(x) < 1e-5 and abs(y) < 1e-5 and abs(yaw) < 1e-5:
            continue                                    # all-zero -> no valid pose yet
        return (x, y, yaw)
    return None

    return slam_rpc.start_mapping(slam_type)

def stop_mapping(save_path=None):
    global _map_path
    if save_path:
        _map_path = save_path
    return slam_rpc.stop_mapping(save_path or _map_path)

def relocate():
    pose = current_pose()
    if pose is None:
        raise RuntimeError("No valid SLAM pose to relocate from.")
    return slam_rpc.init_pose(*pose, address=_map_path)

Named poses are only meaningful once localization is valid, so they are captured from current_pose() (the live relocated pose) and persisted to a small JSON file, not kept only in memory.

_points_path = Path("slam_points.json")

def _load_points():
    return json.loads(_points_path.read_text()) if _points_path.exists() else {}

def _save_points(points):
    tmp = _points_path.with_suffix(".tmp")
    tmp.write_text(json.dumps(points, indent=2))
    os.replace(tmp, _points_path)

def add_point(point_name):
    pose = current_pose()
    if pose is None:
        raise RuntimeError("No valid SLAM pose; relocate first.")
    points = _load_points()
    points[point_name] = pose
    _save_points(points)
    return pose

def remove_point(point_name):
    points = _load_points()
    points.pop(point_name, None)
    _save_points(points)

Subscribe to the live SLAM point-cloud topics and periodically save a 2-D scatter snapshot to disk during mapping; view_map() just returns the newest saved snapshot.

SLAM_POINT_TOPICS = [
    "rt/unitree/slam_mapping/points",
    "rt/unitree/slam_relocation/points",
    "rt/unitree/slam_relocation/global_map",
    "rt/unitree/slam_relocation/web_points",
]
_cloud_subs = [(topic, Latest(topic, PointCloud2_)) for topic in SLAM_POINT_TOPICS]
_snapshots_dir = Path("slam_snapshots"); _snapshots_dir.mkdir(exist_ok=True)

def _decode_xy(msg):
    fields = {str(f.name).lower(): f for f in msg.fields}
    x_off, y_off = fields["x"].offset, fields["y"].offset
    point_step, width, height, raw = msg.point_step, msg.width, msg.height, bytes(msg.data)
    total = min(width * height, len(raw) // max(1, point_step))
    xs, ys = [], []
    for i in range(total):
        base = i * point_step
        xs.append(struct.unpack_from("<f", raw, base + x_off)[0])
        ys.append(struct.unpack_from("<f", raw, base + y_off)[0])
    return xs, ys

def capture_map_snapshot():
    best = max(_cloud_subs, key=lambda pair: pair[1].timestamp)
    topic, latest = best
    if latest.message is None:
        raise RuntimeError("No point-cloud message yet; check the SLAM/LiDAR topics are live.")
    xs, ys = _decode_xy(latest.message)
    plt.figure(figsize=(5, 5)); plt.scatter(xs, ys, s=1); plt.gca().set_aspect("equal")
    path = _snapshots_dir / f"snapshot_{int(time.time())}.png"
    plt.savefig(path); plt.close()
    return path

def view_map():
    snapshots = sorted(_snapshots_dir.glob("snapshot_*.png"))
    return snapshots[-1] if snapshots else None
